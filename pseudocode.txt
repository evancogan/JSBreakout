  start main loop (requestAnimationFrame or Clock)

main loop(dt):
  handle_input()            # mouse/touch -> paddle.x (clamp)
  ball.pos += ball.velocity * dt  # or per-frame step if simpler
  check_wall_collisions()
  if ball.y < bottom:
    lives -= 1
    if lives == 0: game_over()
    else: reset_ball_and_paddle()
  if circle_rect_collision(ball, paddle):
    handle_paddle_collision()   # invert/reflect & optionally modify vx by hit offset
  for each brick in bricks:
    if circle_rect_collision(ball, brick):
      remove brick, score += N
      handle_brick_collision()  # reflect velocity (see below)
  if bricks.empty(): you_win()
  render()  # draw ball, paddle, bricks, UI (score/lives)
